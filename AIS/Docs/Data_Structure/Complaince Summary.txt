create or replace view v_rpt_db_p_compaince_summary_region_total as
select  m.autid as Region_id, m.entitytypedesc as Region, m.audited_by_enitity as GM_ID,
    sum(case when f.para_status = 8 then 1 else 0 end) as Total_para,
       sum(case when f.com_cycle != 0 and f.para_status = 8 then 1 else 0 end) as Total_Comp,
              sum(case when f.com_stage = 21 and f.entity_type_id in (6,28) and f.para_status = 8  then 1 else 0 end )  as AT_reporting,
              sum(case when f.com_stage in (2,43,44) and f.para_status = 8 then 1 else 0 end ) as Under_consideration,
       sum (case when f.com_status in (12,15,18) and f.para_status = 8 then 1 else 0 end) as Rejected,
       sum (case when f.com_status = 16 and f.para_status != 8 then 1 else 0 end) as settled

  from AIS_T_AU_POST_COMPLIANCE f
      inner join t_auditee_ent_types m
      on f.entity_type_id = m.autid
      --where exists (select 'z' from ais_t_au_post_compliance_history h where h.com_id = f.com_id and trunc(h.comment_on) > '26-Jun-2024')
       group by m.autid, m.entitytypedesc, m.audited_by_enitity
       order by m.autid
;

create or replace view v_rpt_db_p_compaince_details_branch as
select  e.entity_id as Region_id, e.name as Region, e.auditby_id as GM_ID,
        m.autid,
        (select count(c.com_id)  from AIS_T_AU_POST_COMPLIANCE c where c.entity_id = e.entity_id and c.para_status = 8) as Total_para,
       sum(case when f.com_cycle != 0  and f.para_status = 8 then 1 else 0 end) as Total_Comp,
              sum(case when f.com_stage = 21  and f.entity_type_id in (6,28) and f.para_status = 8 then 1 else 0 end )  as AT_reporting,
              sum(case when f.com_stage in (2,43,44) and f.para_status = 8 then 1 else 0 end ) as Under_consideration,
       sum (case when f.com_status in (12,15,18) and f.para_status = 8 then 1 else 0 end) as Rejected,
       sum (case when f.com_status = 16 and f.para_status != 8 then 1 else 0 end) as settled

  from AIS_T_AU_POST_COMPLIANCE f
      inner join t_auditee_ent_types m
      on f.entity_type_id = m.autid
      inner join t_auditee_entities e
      on f.entity_id = e.entity_id
--where exists (select 'z' from ais_t_au_post_compliance_history h where h.com_id = f.com_id and trunc(h.comment_on) > '26-Jun-2024')

       group by e.entity_id, e.name, e.auditby_id, m.autid
       order by m.autid
;

create or replace view v_rpt_db_p_compaince_details as
select  e.entity_id as Region_id, e.name as Region, e.auditby_id as GM_ID,
        m.autid,
        sum(case when f.para_status = 8 then 1 else 0 end ) as Total_para,
       sum(case when f.com_cycle != 0 and f.para_status = 8 then 1 else 0 end) as Total_Comp,
              sum(case when f.com_stage = 21 and f.entity_type_id in (6,28) and f.para_status = 8 then 1 else 0 end )  as AT_reporting,
              sum(case when f.com_stage in (43,44,2,6,7,9) and f.para_status = 8 then 1 else 0 end ) as Under_consideration,
       sum (case when f.com_status in (12,15,18) and f.para_status = 8 then 1 else 0 end) as Rejected,
       sum (case when f.com_status = 16 and f.para_status != 8 then 1 else 0 end) as settled

  from AIS_T_AU_POST_COMPLIANCE f
      inner join t_auditee_ent_types m
      on f.entity_type_id = m.autid
      inner join t_auditee_entities e
      on f.entity_id = e.entity_id
       group by e.entity_id, e.name, e.auditby_id, m.autid
       order by m.autid;
